✓ Connected to Supabase

All changes are saved to the cloud and synced to other devices in real time.

Step 1 — Create a free Supabase project
  1. Go to supabase.com and sign up (free).
  2. Click New Project, name it "haykal-books", choose a region near Lebanon.
  3. Wait ~2 minutes for setup to finish.
  4. Go to Project Settings → API and copy your Project URL and anon/public key.
Step 2 — Create the tables (SQL Editor → paste → Run)
create table if not exists schools (
  id text primary key, name text not null, region text, notes text
);
create table if not exists books (
  id text primary key, title text not null, subject text,
  publisher text, isbn text, school text, grade text,
  year text, edition text, price numeric default 0,
  cost numeric default 0, required text, notes text
);
create table if not exists orders (
  id text primary key, customer_name text not null,
  phone text, parent text, school_id text, grade text,
  year text, status text default 'active',
  items jsonb default '[]', discount numeric default 0,
  deposit numeric default 0, final_total numeric default 0,
  remaining numeric default 0, notes text,
  notified boolean default false, notified_at timestamptz,
  created_at timestamptz default now()
);
-- Run if upgrading:
alter table orders add column if not exists notified boolean default false;
alter table orders add column if not exists notified_at timestamptz;
-- Enable real-time:
alter publication supabase_realtime add table schools;
alter publication supabase_realtime add table books;
alter publication supabase_realtime add table orders;
